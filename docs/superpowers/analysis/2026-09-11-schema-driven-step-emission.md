# Schema-Driven Step Emission Architecture (Phase 5c Vision)

## User's Insight

Schema in Lua enables **compositional, type-driven step generation**:

Instead of:
```python
if config.boot:
    if config.boot.kernel:
        # emit kernel steps
    if config.boot.loader:
        # emit loader steps
if config.hardware:
    if config.hardware.pipewire:
        # emit pipewire steps
# ... 20 more if blocks ...
```

We want:
```lua
-- Each section defines: schema + step_generator
local boot = require("sections/boot")     -- Knows its schema, emits its steps
local hardware = require("sections/hardware") -- Knows its schema, emits its steps
local locale = require("sections/locale")   -- Knows its schema, emits its steps

-- Planner just composes them:
function emit_all_steps(config)
    local steps = {}
    for section_name, section_module in pairs(SECTIONS) do
        if config[section_name] then
            -- Section module validates against its schema
            -- Section module emits its steps
            -- No hardcoded conditionals in planner
            local section_steps = section_module.emit_steps(config[section_name])
            table.insert(steps, section_steps)
        end
    end
    return steps
end
```

## This Requires Lua-Primary Schema

Each component (boot, hardware, locale, packages, etc.) needs:

```lua
-- sections/boot.lua
local boot_schema = {
    type = "dict",
    required = false,
    fields = {
        kernel = {
            type = "dict",
            required = false,
            fields = {
                package = { type = "string", default = "linux" },
                modules = { type = "list", required = false }
            }
        },
        loader = {
            type = "dict",
            required = false,
            fields = {
                type = { type = "string", enum = {"systemd-boot", "grub"} },
                timeout = { type = "number", default = 10 }
            }
        }
    }
}

function emit_steps(boot_config)
    -- Module validates boot_config against boot_schema
    if not boot_schema:validate(boot_config) then
        error("Invalid boot config")
    end
    
    local steps = {}
    
    -- Emit kernel steps based on schema defaults and user config
    if boot_config.kernel then
        local kernel_package = boot_config.kernel.package or boot_schema.fields.kernel.fields.package.default
        table.insert(steps, {type="package", name=kernel_package})
        
        if boot_config.kernel.modules then
            for _, module in ipairs(boot_config.kernel.modules) do
                table.insert(steps, {type="kernel_module", name=module})
            end
        end
    end
    
    -- Emit loader steps
    if boot_config.loader then
        local loader_type = boot_config.loader.type or boot_schema.fields.loader.fields.type.default
        table.insert(steps, {type="bootloader", loader=loader_type})
    end
    
    return steps
end

return {
    schema = boot_schema,
    emit_steps = emit_steps
}
```

## Benefits of Schema-Driven Composition

1. **No nested conditionals in planner** — Each section handles its own logic
2. **Reusable validation** — Schema validates before step generation
3. **Self-documenting** — Schema shows what each section expects
4. **Easier to extend** — Add new section = new module with schema + emit_steps
5. **Type safety** — Schema enforces types, defaults, enums, etc.
6. **Composable** — Steps are generated by composition, not coordination
7. **Easier to replace Python** — Compiled language can read Lua schemas and compose
8. **Custom sections** — Users can add new sections with their own schemas

### Step 2: Move Components to Lua with Schemas

```lua
-- src/kod/sections/boot.lua
local Schema = require("schema")

local boot = {
    schema = Schema.boot,
    
    function emit_steps(config, distro)
        local steps = {}
        if not config then return steps end
        
        -- Kernel steps
        if config.kernel then
            local pkg = config.kernel.package or "linux"
            table.insert(steps, {
                type = "package",
                package = pkg,
                manager = distro
            })
            
            if config.kernel.modules then
                for _, mod in ipairs(config.kernel.modules) do
                    table.insert(steps, {
                        type = "kernel_module",
                        module = mod
                    })
                end
            end
        end
        
        -- Loader steps
        if config.loader then
            local loader_type = config.loader.type or "systemd-boot"
            table.insert(steps, {
                type = "bootloader",
                bootloader = loader_type,
                timeout = config.loader.timeout or 10
            })
        end
        
        return steps
    end
}

return boot
```

### Step 3: Planner Becomes Compositional

```lua
-- src/kod/lib/planner.lua
local Schema = require("schema")

-- Load all section modules
local sections = {
    boot = require("sections/boot"),
    hardware = require("sections/hardware"),
    locale = require("sections/locale"),
    network = require("sections/network"),
    users = require("sections/users"),
    -- ...
}

function plan_install(config, distro)
    local plan = {
        metadata = {
            distro = distro,
            mode = "install"
        },
        steps = {}
    }
    
    -- Compose steps from all sections
    for section_name, section_module in pairs(sections) do
        if config[section_name] then
            -- Each section knows its schema and how to emit steps
            local section_steps = section_module.emit_steps(config[section_name], distro)
            for _, step in ipairs(section_steps) do
                table.insert(plan.steps, step)
            end
        end
    end
    
    return plan
end

return {
    plan_install = plan_install,
    sections = sections,
    schema = Schema
}
```

    steps := []Step{}

This works because:
- Schema is in Lua (not Python)
- Step emission is in Lua (not Python)
- Compiled language just orchestrates, doesn't duplicate logic
